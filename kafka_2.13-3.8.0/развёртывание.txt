# 1. Генерация cluster ID
CLUSTER_ID=$(bin/kafka-storage.sh random-uuid)
echo $CLUSTER_ID

# 2. Форматирование хранилища
bin/kafka-storage.sh format -t $CLUSTER_ID -c config/kraft/server.properties

# 3. Запуск брокера
bin/kafka-server-start.sh config/kraft/server.properties


# Создание топика
bin/kafka-topics.sh --create \
  --topic user.events \
  --bootstrap-server localhost:9092 \
  --partitions 1 \
  --replication-factor 1


# Проверка, что топик создан
bin/kafka-topics.sh --list --bootstrap-server localhost:9092

# Тест: отправка сообщения
echo '{"userId":"123","username":"test"}' | \
  bin/kafka-console-producer.sh --topic user.events --bootstrap-server localhost:9092

# Тест: чтение сообщения
bin/kafka-console-consumer.sh \
  --topic user.events \
  --from-beginning \
  --bootstrap-server localhost:9092


# очистка топика
# Удаление и пересоздание — быстро и надёжно
bin/kafka-topics.sh --delete --topic user.events --bootstrap-server localhost:9092
bin/kafka-topics.sh --create --topic user.events --bootstrap-server localhost:9092 --partitions 1 --replication-factor 1
